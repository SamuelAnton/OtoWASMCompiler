    ;;; Integer methods

    ;;;;; Gets value from integer object
    (func $get_integer_value (param $obj i32) (result i32)
        (i32.load offset=8 (local.get $obj))
    )

    ;;;;; Creates integer object
    (func $create_integer (param $value i32) (result i32)
        (local $obj i32)
        
        ;;;;;; Memory allocation (12 bytes)
        (local.set $obj (call $allocate (i32.const 12)))
        
        ;;;;;; [0-4] Method table pointer (0 for integer)
        (i32.store (local.get $obj) (i32.const 0))
        
        ;;;;;; [4-8] Type ID (0 for integer)
        (i32.store offset=4 (local.get $obj) (i32.const 0))
        
        ;;;;;; [8-12] Integer value from parameters
        (i32.store offset=8 (local.get $obj) (local.get $value))
        
        ;;;;;; Return pointer to object in heap
        (local.get $obj)
    )

        ;;;;; Value (f64)
        (f64.store offset=8 (local.get $obj) (local.get $value))
        
        (local.get $obj)
    )

    ;;; Boolean methods
    ;;;; helper functions
    (func $create_boolean (param $value i32) (result i32)
        (local $obj i32)
        (local.set $obj (call $allocate (i32.const 12)))
        
        ;; VTABLE pointer для Boolean
        (i32.store (local.get $obj) (i32.const 228))
        
        ;; Type ID (2 для Boolean)
        (i32.store offset=4 (local.get $obj) (i32.const 2))
        
        ;; Значение (0 = false, 1 = true)
        (i32.store offset=8 (local.get $obj) (local.get $value))
        
        (local.get $obj)
    )

    (func $create_true (result i32)
        (i32.const 1)
        (call $create_boolean)
    )
