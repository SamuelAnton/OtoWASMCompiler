class Animal is
    method sound() => 1
end

class Dog extends Animal is
    method sound() => 2
end

class Polymorphism is
    method polymorphism(a : Animal) is
        a.sound();
    end
end

class Tester is
    this() is
        var x : Animal();
        var y : Dog();
        var z : Polymorphism();
        z.polymorphism(x);
        z.y.polymorphism(y);
    end
end
