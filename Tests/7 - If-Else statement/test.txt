class Tester is
    method test(flag: Boolean) : Integer is
        var x : Integer(1);
        if flag then
            x :=  Integer(5);
        else
            x := Integer(2);
        end
        return x;
    end
end

class Main is
    method main() : Integer is
        var t : Tester();
        return t.test(Boolean(false));
    end
end
