  (func $List.append (param $this i32) (param $value i32) (result i32)
    (local $length i32)
    (local $capacity i32)
    (local $data_ptr i32)
    (local $new_data_ptr i32)
    (local $i i32)
    
    ;;; Load current state
    local.get $this
    i32.load offset=8
    local.set $length
    local.get $this
    i32.load offset=12
    local.set $capacity
    local.get $this
    i32.load offset=16
    local.set $data_ptr
    
    ;;; Check if we need to resize
    local.get $length
    local.get $capacity
    i32.ge_s
    if
      ;;; Double the capacity
      local.get $capacity
      i32.const 2
      i32.mul
      local.set $capacity
      
      ;;; Allocate new data array
      local.get $capacity
      i32.const 4
      i32.mul
      call $allocate
      local.set $new_data_ptr
      
      ;;; Copy old data
      i32.const 0
      local.set $i
      loop $copy_loop
        local.get $i
        local.get $length
        i32.lt_s
        if
          local.get $new_data_ptr
          local.get $i
          i32.const 4
          i32.mul
          i32.add
          local.get $data_ptr
          local.get $i
          i32.const 4
          i32.mul
          i32.add
          i32.load
          i32.store
          local.get $i
          i32.const 1
          i32.add
          local.set $i
          br $copy_loop
        end
      end
      
      ;;; Update list state
      local.get $this
      local.get $capacity
      i32.store offset=12
      local.get $this
      local.get $new_data_ptr
      i32.store offset=16
      local.get $new_data_ptr
      local.set $data_ptr
    end
    
    ;;; Append the value
    local.get $data_ptr
    local.get $length
    i32.const 4
    i32.mul
    i32.add
    local.get $value
    i32.store
    
    ;;; Update length and size
    local.get $this
    local.get $length
    i32.const 1
    i32.add
    i32.store offset=8
    local.get $this
    local.get $length
    i32.const 1
    i32.add
    i32.store offset=20
    
    ;;; Return the list for chaining
    local.get $this
  )
