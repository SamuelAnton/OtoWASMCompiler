    ;;;;; Creates integer object
    (func $create_integer (param $value i32) (result i32)
        (local $obj i32)
        
        ;;;;;; Memory allocation (12 bytes)
        (local.set $obj (call $allocate (i32.const 12)))
        
        ;;;;;; [0-4] Method table pointer (0 for integer)
        (i32.store (local.get $obj) (i32.const 0))
        
        ;;;;;; [4-8] Type ID (0 for integer)
        (i32.store offset=4 (local.get $obj) (i32.const 0))
        
        ;;;;;; [8-12] Integer value from parameters
        (i32.store offset=8 (local.get $obj) (local.get $value))
        
        ;;;;;; Return pointer to object in heap
        (local.get $obj)
    )

    (func $create_real (param $value f64) (result i32)
        (local $obj i32)
        (local.set $obj (call $allocate (i32.const 16)))
        
        ;;;;; VTABLE pointer for Real
        (i32.store (local.get $obj) (i32.const 112))
        
        ;;;;; Type ID (1 for Real)
        (i32.store offset=4 (local.get $obj) (i32.const 1))
        
        ;;;;; Value (f64)
        (f64.store offset=8 (local.get $obj) (local.get $value))
        
        (local.get $obj)
    )

    ;;;;; Creates array object
    (func $create_array (param $length i32) (result i32)
        (local $arr i32)
        (local $total_size i32)
        (local $i i32)
        (local $data_start i32)
        
        ;; Рассчитываем общий размер: заголовок (12 байт) + данные (length * 4 байта)
        (local.set $total_size 
            (i32.add (i32.const 12) (i32.mul (local.get $length) (i32.const 4)))
        )
        
        ;; Выделяем память
        (local.set $arr (call $allocate (local.get $total_size)))
        
        ;; Инициализируем поля массива
        (i32.store (local.get $arr) (global.get $ARRAY_VTABLE))  ;; vtable pointer
        (i32.store offset=4 (local.get $arr) (i32.const 3))      ;; type_id = 3 (Array)
        (i32.store offset=8 (local.get $arr) (local.get $length)) ;; длина массива
        
        ;; Инициализируем элементы нулями (null pointers)
        (local.set $data_start (i32.add (local.get $arr) (i32.const 12)))
        (local.set $i (i32.const 0))
        (loop $init_loop
            (i32.store 
                (i32.add (local.get $data_start) (i32.mul (local.get $i) (i32.const 4)))
                (i32.const 0)  ;; null pointer
            )
            (local.set $i (i32.add (local.get $i) (i32.const 1)))
            (br_if $init_loop (i32.lt_u (local.get $i) (local.get $length)))
        )
        
        (local.get $arr)
    )

    ;;;;; Gets array length
    (func $get_array_length (param $arr i32) (result i32)
        (i32.load offset=8 (local.get $arr))
    )

    ;;;;; Gets element from array
    (func $get_array_element (param $arr i32) (param $index i32) (result i32)
        ;; Проверка границ (опционально, но рекомендуется)
        (if (i32.ge_u (local.get $index) (i32.load offset=8 (local.get $arr)))
            (then
                ;; Выход за границы - возвращаем null
                (return (i32.const 0))
            )
        )
        
        (i32.load 
            (i32.add 
                (i32.add (local.get $arr) (i32.const 12))
                (i32.mul (local.get $index) (i32.const 4))
            )
        )
    )

    ;;;;; Sets element in array
    (func $set_array_element (param $arr i32) (param $index i32) (param $value i32) (result i32)
        ;; Проверка границ
        (if (i32.ge_u (local.get $index) (i32.load offset=8 (local.get $arr)))
            (then
                ;; Выход за границы - просто возвращаем массив без изменений
                (return (local.get $arr))
            )
        )
        
        (i32.store 
            (i32.add 
                (i32.add (local.get $arr) (i32.const 12))
                (i32.mul (local.get $index) (i32.const 4))
            )
            (local.get $value)
        )
        
        (local.get $arr)
    )

    ;;; Ensures the List has enough capacity, resizing if necessary
    (func $list_ensure_capacity (param $list i32) (param $min_capacity i32)
        (local $current_capacity i32)
        (local $new_capacity i32)
        (local $current_data i32)
        (local $new_data i32)
        (local $i i32)
        
        ;; Get current capacity
        (local.set $current_capacity (call $list_get_capacity (local.get $list)))
        
        ;; If current capacity is sufficient, return
        (if (i32.ge_u (local.get $current_capacity) (local.get $min_capacity))
            (then (return))
        )
        
        ;; Calculate new capacity (double until sufficient)
        (local.set $new_capacity (local.get $current_capacity))
        (if (i32.eq (local.get $new_capacity) (i32.const 0))
            (then (local.set $new_capacity (i32.const 4)))
        )
        
        (loop $calculate_capacity
            (if (i32.lt_u (local.get $new_capacity) (local.get $min_capacity))
                (then
                    (local.set $new_capacity (i32.mul (local.get $new_capacity) (i32.const 2)))
                    (br $calculate_capacity)
                )
            )
        )
        
        ;; Create new array with new capacity
        (local.set $new_data (call $create_array (local.get $new_capacity)))
        
        ;; Copy existing elements if any
        (local.set $current_data (call $list_get_data (local.get $list)))
        (if (i32.ne (local.get $current_data) (i32.const 0))
            (then
                (local.set $i (i32.const 0))
                (loop $copy_loop
                    (if (i32.lt_u (local.get $i) (call $list_get_size (local.get $list)))
                        (then
                            ;; Copy element from old array to new array
                            (drop (call $set_array_element
                                (local.get $new_data)
                                (local.get $i)
                                (call $get_array_element (local.get $current_data) (local.get $i))
                            ))
                            (local.set $i (i32.add (local.get $i) (i32.const 1)))
                            (br $copy_loop)
                        )
                    )
                )
            )
        )
        
        ;; Update List fields
        (call $list_set_data (local.get $list) (local.get $new_data))
        (call $list_set_capacity (local.get $list) (local.get $new_capacity))
    )

    (func $List.constructor2 (type $list_constructor2)
        (param $this i32) (param $element i32) (param $count i32) (result i32)
        (local $i i32)
        (local $data i32)
        
        ;; Ensure capacity
        (call $list_ensure_capacity (local.get $this) (local.get $count))
        
        ;; Get data array
        (local.set $data (call $list_get_data (local.get $this)))
        
        ;; Fill with copies of the element
        (local.set $i (i32.const 0))
        (loop $fill_loop
            (if (i32.lt_u (local.get $i) (local.get $count))
                (then
                    (call $set_array_element
                        (local.get $data)
                        (local.get $i)
                        (local.get $element)
                    )
                    (local.set $i (i32.add (local.get $i) (i32.const 1)))
                    (br $fill_loop)
                )
            )
        )
        
        ;; Update size
        (call $list_set_size (local.get $this) (local.get $count))
        
        (local.get $this)
    )

    ;;; Append element to list
    (func $List.append (type $list_append)
        (param $this i32) (param $element i32) (result i32)
        (local $size i32)
        (local $data i32)
        
        ;; Get current size and ensure capacity
        (local.set $size (call $list_get_size (local.get $this)))
        (call $list_ensure_capacity (local.get $this) (i32.add (local.get $size) (i32.const 1)))
        
        ;; Get data array
        (local.set $data (call $list_get_data (local.get $this)))
        
        ;; Add element at the end
        (drop (call $set_array_element
            (local.get $data)
            (local.get $size)
            (local.get $element)
        ))
        
        ;; Update size
        (call $list_set_size (local.get $this) (i32.add (local.get $size) (i32.const 1)))
        
        (local.get $this)
    )

    ;;; Get first element (head)
    (func $List.head (type $list_head)
        (param $this i32) (result i32)
        (local $size i32)
        
        (local.set $size (call $list_get_size (local.get $this)))
        
        ;; Check if list is empty
        (if (i32.eq (local.get $size) (i32.const 0))
            (then
                ;; Return null for empty list (you might want to throw an error instead)
                (return (i32.const 0))
            )
        )
        
        ;; Return first element
        (call $get_array_element
            (call $list_get_data (local.get $this))
            (i32.const 0)
        )
    )

    ;;; Get tail (list without first element)
    (func $List.tail (type $list_tail)
        (param $this i32) (result i32)
        (local $size i32)
        (local $new_list i32)
        (local $data i32)
        (local $new_data i32)
        (local $i i32)
        
        (local.set $size (call $list_get_size (local.get $this)))
        
        ;; If list is empty or has only one element, return empty list
        (if (i32.le_u (local.get $size) (i32.const 1))
            (then
                (return (call $create_list))
            )
        )
        
        ;; Create new list
        (local.set $new_list (call $create_list))
        
        ;; Ensure capacity for new list
        (call $list_ensure_capacity (local.get $new_list) (i32.sub (local.get $size) (i32.const 1)))
        
        ;; Get data arrays
        (local.set $data (call $list_get_data (local.get $this)))
        (local.set $new_data (call $list_get_data (local.get $new_list)))
        
        ;; Copy all elements except first
        (local.set $i (i32.const 1))
        (loop $copy_loop
            (if (i32.lt_u (local.get $i) (local.get $size))
                (then
                    (drop (call $set_array_element
                        (local.get $new_data)
                        (i32.sub (local.get $i) (i32.const 1))
                        (call $get_array_element (local.get $data) (local.get $i))
                    ))
                    (local.set $i (i32.add (local.get $i) (i32.const 1)))
                    (br $copy_loop)
                )
            )
        )
        
        ;; Set size of new list
        (call $list_set_size (local.get $new_list) (i32.sub (local.get $size) (i32.const 1)))
        
        (local.get $new_list)
    )

    ;;; Get size of list
    (func $List.size (type $list_size)
        (param $this i32) (result i32)
        ;; Return size as Integer object
        (call $list_get_size (local.get $this))
        (call $create_integer)

        ;; Return Boolean object
        (call $list_get_size (local.get $this))
        (i32.eqz)
        (call $create_boolean)
    )

    (func $allocate (param $size i32) (result i32)
        (local $ptr i32)
        
        (local.set $ptr (global.get $heap_ptr))
        
        (global.set $heap_ptr
        (i32.add (global.get $heap_ptr) (local.get $size))
        )
        
        (local.get $ptr)
    )

    ;; Get method from object and method index
    ;; Params: $obj i32, $method_index i32
    ;; Returns: function index in table
    (func $get_method (param $obj i32) (param $method_index i32) (result i32)
        (local $vtable i32)
        
        ;; Get VTable pointer from object
        (local.set $vtable (i32.load (local.get $obj)))
        
        ;; Skip type ID (4 bytes) and method count (4 bytes)
        ;; Then get method at index: vtable + 8 + (method_index * 4)
        (i32.load 
            (i32.add 
                (i32.add (local.get $vtable) (i32.const 8))
                (i32.mul (local.get $method_index) (i32.const 4))
            )
        )
    )
